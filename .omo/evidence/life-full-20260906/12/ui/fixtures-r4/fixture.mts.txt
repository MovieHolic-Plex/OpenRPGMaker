import { loadProjectFromLegacyDb, saveProjectToLegacyDb } from '@/project/legacyDbProjectSync';

import { resolvePlayerBody, playerBodyRect, playerPassageRect } from '@/project/playerFootprint';

const out = '.omo/evidence/life-full-20260906/12/ui/fixtures-r4';
const ids = { arena: 'rpg-zzu-life-full-p4-t12-arena-st-01a07ba6', lastExit: 'rpg-zzu-life-full-p4-t12-last-exit-st-01a07ba6' };
const hash = (s: string) => createHash('sha256').update(s).digest('hex');
const write = (name: string, data: unknown) => writeFileSync(`${out}/${name}`, JSON.stringify(data, null, 2)+'\n');
const env: Record<string,string> = {};
for (const file of ['.env','.env.local']) if (existsSync(file)) for (const line of readFileSync(file,'utf8').split(/\r?\n/)) {
  const m = line.match(/^\s*([A-Z0-9_]+)\s*=\s*(.*?)\s*$/); if(m) env[m[1]] = m[2].replace(/^["']|["']$/g,'');
}
const config = (projectId: string) => ({ projectId, url: (process.env.VITE_LEGACY_DB_URL ?? env.VITE_LEGACY_DB_URL ?? '').replace(/\/$/,''), anonKey: process.env.VITE_LEGACY_DB_ANON_KEY ?? env.VITE_LEGACY_DB_ANON_KEY ?? '' });
assert(config(ids.arena).url && config(ids.arena).anonKey, 'DB credentials required');
const mode = process.argv.at(-1);
const requests: unknown[] = [];
const originalFetch = globalThis.fetch;
globalThis.fetch = async (input, init) => {
  const url = new URL(input instanceof Request ? input.url : String(input));
  const result = await originalFetch(input, init);
  requests.push({method:init?.method ?? 'GET', path:url.pathname, query:url.search, status:result.status});
  write(`${mode}.requests.json`,requests);
  return result;
};
function inputProject(kind: keyof typeof ids): Project {
  const project = {
    ...deserialize(readFileSync('.omo/evidence/life-full-20260906/12/ui/producer-r3/native/fixture.json','utf8')),
    // Preserve the r3 wire's legacy extra name field; identity is the remote project ID.
    name: `Task12 r4 ${kind} st_01a07ba6`,
  };
  project.startPos = kind === 'arena' ? {x:8,y:8} : {x:1,y:2};
  project.maps[project.startMapId].events = [];
  const shed = project.database.farmBuildingTypes!.find(t => t.id === 'shed')!;
  project.database.farmBuildingTypes = [{...shed, levels:[{...shed.levels[0], footprint:{width:1,height:1}, cost:{gold:10,items:[{itemId:'item_potion',count:2}]}}]}];
  const table = project.database.homeDecorationTypes!.find(t => t.id === 'table')!;
  project.database.homeDecorationTypes = [{...table, footprint:{width:2,height:1}, allowedOrientations:['down','left']}];
  project.session.farmBuildingPlacements = kind === 'arena' ? [] : [
    {instanceId:'r4-block-up',typeId:'shed',mapId:project.startMapId,x:1,y:1,orientation:'down',level:1},
    {instanceId:'r4-block-right',typeId:'shed',mapId:project.startMapId,x:3,y:2,orientation:'down',level:1},
  ];
  project.session.homeDecorationPlacements = [];
  assert(!('farmPlots' in project.session));
  return project;
}
function prove(project: Project, kind: keyof typeof ids) {
  const serializedBefore = serialize(project);
  const session = startSession(project,1204);
  const initial = structuredClone(session);
  const mapId = project.startMapId, map = project.maps[mapId];
  const body = resolvePlayerBody(project,session);
  assert.deepEqual(body,{footprint:{width:3,height:3},passRows:1});
  assert.deepEqual(session.farmPlots,{});
  assert.equal(map.events.length,0);
  assert.deepEqual(session.placeables,{});
  const readLive = () => ({player:{mapId:session.currentMapId,x:session.x,y:session.y,...body},npcs:[]});
  const playerRects = () => ({body:playerBodyRect(body,session.x,session.y),passage:playerPassageRect(body,session.x,session.y)});
  const walk = (direction: Dir) => {
    const [dx,dy] = {up:[0,-1],down:[0,1],left:[-1,0],right:[1,0]}[direction];
    const x = session.x+dx, y = session.y+dy;
    const passage = playerPassageRect(body,x,y);
    const terrain = canMoveFootprint(project,map,session.x,session.y,body.footprint,x,y,body.passRows);
    const persistent = isSpatialPlacementBlocking(project,session,mapId,passage);
    return {direction,x,y,passage,terrain,persistent,accepted:terrain && !persistent};
  };
  let details: unknown;
  if (kind === 'arena') {
    assert.deepEqual({x:session.x,y:session.y},{x:8,y:8});
    assert.deepEqual(session.farmBuildingPlacements,{});
    assert.deepEqual(session.homeDecorationPlacements,{});
    const table = project.database.homeDecorationTypes!.find(t=>t.id==='table')!;
    assert.deepEqual(table.footprint,{width:2,height:1});
    assert.deepEqual(table.allowedOrientations,['down','left']);
    assert.equal(table.blocksMovement,true);
    const target = adjacentSpatialPosition(readLive().player,'down',table.footprint,'down');
    assert.deepEqual(target,{mapId,x:7,y:9,orientation:'down'});
    const staticAccepted = canOccupySpatialFootprint(project,session,target,table.footprint);
    const liveAccepted = canPlaceSpatialFootprint(project,session,target,table.footprint,readLive);
    assert(staticAccepted && liveAccepted);
    const beforeRects = playerRects();
    const placed = placeHomeDecoration(project,session,{...target,instanceId:'r4-probe-table',typeId:'table'},readLive);
    assert.deepEqual(placed,{ok:true});
    assert.equal(session.inventory[table.placementItemId],initial.inventory[table.placementItemId]-1);
    assert.equal(session.gold,initial.gold);
    const afterPlace = structuredClone(session);
    const rotatedTarget = {...target,orientation:'left' as const};
    const rotationAccepted = canPlaceSpatialFootprint(project,session,rotatedTarget,table.footprint,readLive,{kind:'homeDecoration',instanceId:'r4-probe-table'});
    assert(rotationAccepted);
    const rotated = rotateHomeDecoration(project,session,'r4-probe-table','left',readLive);
    assert.deepEqual(rotated,{ok:true});
    assert.deepEqual(session.homeDecorationPlacements!['r4-probe-table'],{...afterPlace.homeDecorationPlacements!['r4-probe-table'],orientation:'left'});
    assert.deepEqual(orientedFootprint(table.footprint,'left'),{width:1,height:2});
    const afterRotate = structuredClone(session);
    const leftStep = walk('left'); assert(leftStep.accepted);
    // MODEL-ONLY setup after public walk-feasibility checks. Not native input proof.
    session.x=leftStep.x; session.y=leftStep.y;
    const moveTarget = adjacentSpatialPosition(readLive().player,'left',table.footprint,'left');
    assert.deepEqual(moveTarget,{mapId,x:5,y:6,orientation:'left'});
    const moveAccepted = canPlaceSpatialFootprint(project,session,moveTarget,table.footprint,readLive,{kind:'homeDecoration',instanceId:'r4-probe-table'});
    assert(moveAccepted);
    const moved = moveHomeDecoration(project,session,'r4-probe-table',mapId,moveTarget.x,moveTarget.y,readLive);
    assert.deepEqual(moved,{ok:true});
    assert.deepEqual(session.homeDecorationPlacements!['r4-probe-table'],{...afterRotate.homeDecorationPlacements!['r4-probe-table'],x:5,y:6});
    assert.deepEqual(structuredClone(session.inventory),afterPlace.inventory); assert.equal(session.gold,afterPlace.gold);
    const expected = structuredClone(afterPlace);
    expected.x=7; expected.y=8;
    expected.homeDecorationPlacements!['r4-probe-table']={...expected.homeDecorationPlacements!['r4-probe-table'],orientation:'left',x:5,y:6};
    assert.deepEqual(structuredClone(session),expected,'no unrelated rotation/move costs or owners');
    details = {table,beforeRects,staticAccepted,liveAccepted,target,placed,afterPlace,rotationAccepted,rotatedTarget,
      rotatedCells:footprintCells(7,9,table.footprint,'left'),rotated,afterRotate,leftStep,afterWalkRects:playerRects(),moveTarget,moveAccepted,moved};
  } else {
    assert.deepEqual({x:session.x,y:session.y},{x:1,y:2});
    assert.deepEqual(Object.keys(session.farmBuildingPlacements!).sort(),['r4-block-right','r4-block-up']);
    const type = project.database.farmBuildingTypes!.find(t=>t.id==='shed')!;
    const fp = type.levels[0].footprint;
    assert.deepEqual(fp,{width:1,height:1});
    const exits = (['up','down','left','right'] as const).map(walk);
    assert.deepEqual(exits.filter(e=>e.accepted).map(e=>e.direction),['down']);
    const target = adjacentSpatialPosition(readLive().player,'down',fp,'down');
    assert.deepEqual(target,{mapId,x:0,y:3,orientation:'down'});
    const staticAccepted = canOccupySpatialFootprint(project,session,target,fp);
    const nonblockingLiveAccepted = canPlaceSpatialFootprint(project,session,target,fp,readLive,undefined,false);
    const blockingLiveAccepted = canPlaceSpatialFootprint(project,session,target,fp,readLive);
    assert(staticAccepted && nonblockingLiveAccepted && !blockingLiveAccepted);
    assert(isPassable(project,map,0,3));
    const input = {...target,instanceId:'r4-probe-last-exit',typeId:'shed'};
    // Separate positive control proves affordability/type validity, never saved as an input.
    const staticSession = structuredClone(session);
    const staticConstruction = placeFarmBuilding(project,staticSession,input);
    assert.deepEqual(staticConstruction,{ok:true});
    assert.equal(staticSession.gold,session.gold-10);
    assert.equal(staticSession.inventory.item_potion,session.inventory.item_potion-2);
    const result = placeFarmBuilding(project,session,input,readLive);
    assert.deepEqual(result,{ok:false,reason:'blocked'});
    assert.deepEqual(session,initial,'full session atomic refusal, not only gold');
    details = {type,playerRects:playerRects(),exits,target,staticAccepted,nonblockingLiveAccepted,blockingLiveAccepted,
      staticConstruction,staticPositiveControlAfter:staticSession,result,fullSessionUnchanged:true};
  }
  assert.equal(serialize(project),serializedBefore,'probe must not mutate authored project');
  return {kind,modelOnlyNotNativeProof:true,initialFacing:'down',mapId,mapSize:{width:map.width,height:map.height},body,
    initial,details,after:session,authoredBeforeSha256:hash(serializedBefore),authoredAfterSha256:hash(serialize(project)),authoredUnchanged:true};
}
if (mode === 'connection') {
  const checks = [];
  for (const projectId of Object.values(ids)) {
    assert.equal(await loadProjectFromLegacyDb(config(projectId)), null, 'ID must not already exist');
    checks.push({projectId, connectionSucceeded:true, absentBeforeAuthoring:true});
  }
  write('connection-proof.json',{checks, credentialsPublished:false});
  console.log(JSON.stringify(checks));
} else if (mode === 'author') {
  assert(existsSync(`${out}/connection-proof.json`),'connection must precede authoring');
  for (const [kind,projectId] of Object.entries(ids) as [keyof typeof ids,string][]) {
    assert.equal(await loadProjectFromLegacyDb(config(projectId)),null,'never overwrite existing content');
    const project = inputProject(kind);
    const authored = serialize(project);
    writeFileSync(`${out}/${kind}.authored.json`,authored);
    const saved = await saveProjectToLegacyDb(project,config(projectId));
    assert.equal(saved.kind,'saved');
    write(`${kind}.save-receipt.json`,{projectId,...saved,project:undefined,authoredSha256:hash(authored)});
    const reloaded = await loadProjectFromLegacyDb(config(projectId)); assert(reloaded);
    const bytes = serialize(reloaded);
    assert.deepEqual(resolvePlayerBody(reloaded),{footprint:{width:3,height:3},passRows:1});
    assert.deepEqual(reloaded.session.farmBuildingPlacements,project.session.farmBuildingPlacements);
    assert.deepEqual(reloaded.session.homeDecorationPlacements,[]);
    assert(!('farmPlots' in reloaded.session));
    writeFileSync(`${out}/${kind}.reloaded.json`,bytes);
    write(`${kind}.reload-receipt.json`,{projectId,saveKind:saved.kind,savedSha256:saved.kind==='saved'?saved.sha256:null,
      authoredSha256:hash(authored),serializedReloadSha256:hash(bytes),playerFootprint:reloaded.system.playerFootprint,
      playerPassRows:reloaded.system.playerPassRows,startPos:reloaded.startPos,mapId:reloaded.startMapId,
      farmBuildingPlacements:reloaded.session.farmBuildingPlacements,homeDecorationPlacements:reloaded.session.homeDecorationPlacements});
    console.log(JSON.stringify({kind,projectId,saved:saved.kind,reloadSha256:hash(bytes)}));
  }
} else if (mode === 'probe' || mode === 'probe-final' || mode === 'probe-certified') {
  for (const [kind,projectId] of Object.entries(ids) as [keyof typeof ids,string][]) {
    const fixture = readFileSync(`${out}/${kind}.reloaded.json`,'utf8');
    const project = await loadProjectFromLegacyDb(config(projectId)); assert(project);
    assert.equal(serialize(project),fixture,'remote input must match immutable fixture');
    const proof = prove(project,kind);
    assert.equal(readFileSync(`${out}/${kind}.reloaded.json`,'utf8'),fixture);
    const after = await loadProjectFromLegacyDb(config(projectId)); assert(after);
    assert.equal(serialize(after),fixture,'remote authored input unchanged after probes');
    write(`${kind}.core-proof.json`,{projectId,...proof,fixtureBeforeSha256:hash(fixture),fixtureAfterSha256:hash(readFileSync(`${out}/${kind}.reloaded.json`,'utf8')),
      remoteAfterSha256:hash(serialize(after)),remoteUnchanged:true});
    console.log(JSON.stringify({kind,projectId,allAssertionsPassed:true,modelOnlyNotNativeProof:true}));
  }
} else throw new Error('Unsupported mode');
