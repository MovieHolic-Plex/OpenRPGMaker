import { mkdir } from "node:fs/promises";
import { expect, test, type Locator, type Page } from "@playwright/test";
import { createBlankProject } from "@/project/defaults";
import type { Command, EventPage, Project } from "@/project/types";
import { seedProjectFromSupabaseCanonical } from "./supabaseProjectSeed";

/** 재설계 전(=main, 3438cacd) 도크의 실제 화면. 보고서 «전» 칸에 쓴다. */
const SHOT_DIR = "output/evidence/event-ai-dock-baseline";

const EXISTING_COMMANDS: readonly Command[] = [
  { kind: "text", body: "나무 상자를 열어본다.", speaker: "" } as Command,
  { kind: "text", body: "잠겨 있다.", speaker: "" } as Command,
];

/** 예전 계약: 모델이 «끼워 넣을 새 명령»만 낸다. */
const INSERTED: readonly Command[] = [
  { kind: "text", body: "나무 상자를 열었다.", speaker: "" } as Command,
  { kind: "wait", ms: 500 } as Command,
  { kind: "setSelfSwitch", key: "A", value: true } as Command,
];

async function mockLlm(page: Page, commands: readonly Command[]): Promise<void> {
  await page.route("**/v1/chat/completions", async (route) => {
    await route.fulfill({
      status: 200,
      contentType: "application/json",
      body: JSON.stringify({
        choices: [{ message: { role: "assistant", content: JSON.stringify(commands) } }],
      }),
    });
  });
}

test.setTimeout(180_000);

test.beforeEach(async ({ page }) => {
  await page.addInitScript(() => {
    window.localStorage.clear();
    window.sessionStorage.clear();
    window.localStorage.setItem("oprn:editor-session-id", "e2e-ai-dock-baseline");
    window.localStorage.setItem("oprn:editor-ui-mode", "standard");
  });
  await mkdir(SHOT_DIR, { recursive: true });
});

function probeProject(): Project {
  const project = createBlankProject();
  const map = project.maps[project.startMapId];
  if (!map) throw new Error("blank project has no start map");
  const commands: Command[] = structuredClone(EXISTING_COMMANDS) as Command[];
  const eventPage: EventPage = {
    id: "p1",
    name: "상자",
    conditions: [],
    graphic: {},
    trigger: { kind: "action" },
    priority: "same",
    movement: { type: "fixed", speed: 3, frequency: 3 },
    commands,
  };
  map.events = [
    { id: "ev_ai_dock", x: 4, y: 4, pages: [eventPage], commands } as (typeof map.events)[number],
  ];
  return project;
}

async function openDock(page: Page): Promise<Locator> {
  const project = probeProject();
  await seedProjectFromSupabaseCanonical(page, project);
  await page.evaluate(async (mapId) => {
    const modalModule = await import("/src/editor/panels/eventEditor/modal.ts");
    modalModule.openEventEditorModal(mapId, "ev_ai_dock");
  }, project.startMapId);
  const editor = page.getByTestId("event-editor-modal");
  await expect(editor).toBeVisible();
  await editor.getByTestId("event-view-toggle-list").click();
  await editor.getByTestId("event-command-quick-ai").click();
  await expect(editor.getByTestId("ai-event-assist")).toHaveJSProperty("open", true);
  return editor;
}

function commandsColumn(editor: Locator): Locator {
  return editor.locator(".event-editor-commands-column");
}

test("before: 도크를 열면 «맨 아래에 넣습니다»라고 말한다", async ({ page }) => {
  await page.setViewportSize({ width: 1440, height: 960 });
  const editor = await openDock(page);
  await expect(editor.getByTestId("ai-event-target")).toHaveText("맨 아래에 이어서 넣습니다");
  await editor.screenshot({ path: `${SHOT_DIR}/before-01-dock-open.png` });
  await commandsColumn(editor).screenshot({ path: `${SHOT_DIR}/before-01b-column.png` });
});

test("before: 초안은 도크 안의 글자 목록이고, 명령 목록과 따로 놓인다", async ({ page }) => {
  await page.setViewportSize({ width: 1440, height: 960 });
  await mockLlm(page, INSERTED);
  const editor = await openDock(page);

  await editor.getByTestId("ai-event-input").fill("상자를 열면 «열었다»를 말하고 잠금 기억을 켠다");
  await editor.getByTestId("ai-event-generate").click();
  await expect(editor.locator(".ai-event-preview-line")).toHaveCount(INSERTED.length);

  await editor.screenshot({ path: `${SHOT_DIR}/before-02-preview.png` });
  await commandsColumn(editor).screenshot({ path: `${SHOT_DIR}/before-02b-column.png` });
  await editor.getByTestId("ai-event-preview").screenshot({ path: `${SHOT_DIR}/before-02c-preview-only.png` });
  console.info(`before chip: ${await editor.getByTestId("ai-event-chip-status").innerText()}`);
});

test("before: 3개를 넣으면 되돌리기를 3번 눌러야 원래대로 온다", async ({ page }) => {
  await page.setViewportSize({ width: 1440, height: 960 });
  await mockLlm(page, INSERTED);
  const editor = await openDock(page);
  const rows = editor.locator(".event-contents-fieldset .cmd-list [data-cmd-path]");

  await editor.getByTestId("ai-event-input").fill("상자를 열면 «열었다»를 말하고 잠금 기억을 켠다");
  await editor.getByTestId("ai-event-generate").click();
  await expect(editor.locator(".ai-event-preview-line")).toHaveCount(INSERTED.length);
  await editor.getByTestId("ai-event-insert").click();
  await expect(rows).toHaveCount(EXISTING_COMMANDS.length + INSERTED.length);
  await commandsColumn(editor).screenshot({ path: `${SHOT_DIR}/before-03-inserted-5rows.png` });

  // 되돌리기 한 번 — 예전 구현은 명령마다 스냅샷을 쌓아 한 줄만 빠진다.
  await editor.getByTestId("event-command-toolbar-undo").click();
  await expect(rows).toHaveCount(EXISTING_COMMANDS.length + INSERTED.length - 1);
  await commandsColumn(editor).screenshot({ path: `${SHOT_DIR}/before-04-undo-once.png` });

  let clicks = 1;
  while ((await rows.count()) > EXISTING_COMMANDS.length && clicks < 10) {
    await editor.getByTestId("event-command-toolbar-undo").click();
    clicks += 1;
  }
  await expect(rows).toHaveCount(EXISTING_COMMANDS.length);
  console.info(`before undo: applied=${INSERTED.length} clicksNeeded=${clicks}`);
});

test("before: «고쳐 달라»고 하면 고치는 대신 아래에 또 붙는다", async ({ page }) => {
  await page.setViewportSize({ width: 1440, height: 960 });
  // 예전 계약대로 «새 명령 한 개»만 낸다 — 고친 문구지만 삽입이므로 원래 줄이 남는다.
  await mockLlm(page, [{ kind: "text", body: "굳게 잠겨 있다. 열쇠가 있어야 한다.", speaker: "" } as Command]);
  const editor = await openDock(page);
  const rows = editor.locator(".event-contents-fieldset .cmd-list [data-cmd-path]");

  await editor.getByTestId("ai-event-input").fill("둘째 줄 대사를 더 단단한 말투로 고쳐 줘");
  await editor.getByTestId("ai-event-generate").click();
  await expect(editor.locator(".ai-event-preview-line")).toHaveCount(1);
  await editor.getByTestId("ai-event-insert").click();

  // 결과: 2줄이 아니라 3줄. 옛 문구가 그대로 남는다.
  await expect(rows).toHaveCount(3);
  await commandsColumn(editor).screenshot({ path: `${SHOT_DIR}/before-05-edit-becomes-duplicate.png` });
  const texts = await rows.allInnerTexts();
  console.info(`before edit-request rows=${rows ? 3 : 0}: ${JSON.stringify(texts.map((t) => t.replace(/\s+/g, " ").slice(0, 40)))}`);
});
